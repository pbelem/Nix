INSTALLATION GUIDE

1 step:
sudo -i
lsblk
cfdisk /dev/sda

2 step:
mkfs.etx4 -L nixos /dev/sda2
mkfs.fat -F 32 -n BOOT /dev/sda1

3 step:
mount /dev/sda2 /mnt
mount --mkdir /dev/sda1 /mnt/boot

4 step:
# use one of these to change the system time
sudo timedatectl set-ntp true
timedatectl set-timezone America/Bahia
date -s "2026-04-28 09:50:00"

5 step:
nixos-generate-config --root /mnt

6 step:
cd /tmp
git clone https://github.com/pbelem/nixos-belem.git
cd /Nix
cp ./* /mnt/etc/nixos
rm /mnt/etc/nixos/installation.txt

7 step:
cd /mnt/etc/nixos
git init
git branch -m main
git add .
git commit -m "system"

7 step:
nixos-install --root /mnt --flake /mnt/etc/nixos#Desktop-NixOS

8 step:
nixos-enter --root /mnt -c 'passwd belem'

9 step:
reboot

10 step:
(q) Quit and do nothing
sudo chown -R belem:users /etc/nixos
nix run home-manager/release-25.11 -- switch --flake /etc/nixos#belem
# or alias = nrhm

11 step:
reboot
super + enter
noctalia-shell & disown

SYSTEM INSTALLED
